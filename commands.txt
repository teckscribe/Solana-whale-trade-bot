# ==============================================================================
# 🐋 WTB Solana Whale Tracker Bot — Systemd Service Commands (Ubuntu 24.04)
# ==============================================================================

# ── Service Architecture ──────────────────────────────────────────────────────
#  • wtb.service        → Main Trading Engine (main.py)
#  • wtb-bot.service    → Telegram Control Bot (telegram_bot.py) [alias: wtb-tg]
#  • wtb-web.service    → FastAPI Web Dashboard (web_server.py)  [alias: wtb-ui]
#  • wtb-ngrok.service  → Ngrok Remote Web Tunnel (ngrok.yml)
# ─────────────────────────────────────────────────────────────────────────────

# 1. Initial Setup (Run Once on VPS)
chmod +x install_services.sh
./install_services.sh

# 2. Check Status of All Services
sudo systemctl status wtb wtb-bot wtb-web wtb-ngrok

# Or compact check:
systemctl is-active wtb wtb-bot wtb-web wtb-ngrok

# 3. Start All Services
sudo systemctl start wtb wtb-bot wtb-web wtb-ngrok

# 4. Stop All Services
sudo systemctl stop wtb wtb-bot wtb-web wtb-ngrok

# 5. Restart All Services
sudo systemctl restart wtb wtb-bot wtb-web wtb-ngrok

# 6. Restart Individual Services
sudo systemctl restart wtb          # Trading engine
sudo systemctl restart wtb-bot      # Telegram bot
sudo systemctl restart wtb-web      # Web dashboard
sudo systemctl restart wtb-ngrok    # Ngrok tunnel

# 7. Live Log Streams
sudo journalctl -u wtb -f           # Live bot engine stream
sudo journalctl -u wtb-bot -f       # Telegram bot stream
sudo journalctl -u wtb-web -f       # Web server stream
sudo journalctl -u wtb-ngrok -f     # Ngrok tunnel stream

# 8. Git Pull & Fast Refresh
git pull origin main
sudo systemctl restart wtb wtb-bot wtb-web wtb-ngrok
