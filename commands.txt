#---------Managing the Main Bot (wtb.service)---------


Start: sudo systemctl start wtb.service
Stop: sudo systemctl stop wtb.service
Restart: sudo systemctl restart wtb.service
View Live Logs: sudo journalctl -u wtb.service -f
Check Status: sudo systemctl status wtb.service


#---------Managing the Telegram Bot (wtb-tg.service)--------
Start: sudo systemctl start wtb-tg.service
Stop: sudo systemctl stop wtb-tg.service
Restart: sudo systemctl restart wtb-tg.service
View Live Logs: sudo journalctl -u wtb-tg.service -f
Check Status: sudo systemctl status wtb-tg.service

#--------Manage NGROK server-----------

sudo systemctl restart wtb-ngrok.service

sudo systemctl enable wtb-ngrok.service
sudo systemctl start wtb-ngrok.service
